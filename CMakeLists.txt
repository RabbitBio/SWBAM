cmake_minimum_required(VERSION 3.13)
include(swbam-sdk/cmake/SunwayToolchain.cmake)
project(SWBAM LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_SWBAM_APP "Build the SWBAM command-line application" ON)
option(BUILD_SWBAM_EXAMPLES "Build the two library examples" ON)
option(SWBAM_BUILD_TESTS "Build the small MPI correctness regression" OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
add_subdirectory(swbam-sdk)
if(BUILD_SWBAM_APP)
    add_subdirectory(swbam-app)
endif()
if(SWBAM_BUILD_TESTS AND BUILD_SWBAM_APP)
    add_subdirectory(tests)
endif()
